Copia este código en tu proyecto de Google Apps Script:

function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    
    // 1. Registrar en la hoja de cálculo
    sheet.appendRow([
      new Date(),
      data.cliente,
      data.email,
      data.ubicacion,
      data.capacidadKWp,
      data.produccionAnualKWh,
      data.consumoAnualKWh,
      data.autosuficienciaPct,
      data.inversionUSD,
      data.roiAnios,
      data.co2Ton,
      data.observaciones
    ]);
    
    // 2. Enviar correo automático
    var mensaje = "Estimado/a " + data.cliente + ",\n\n" +
      "Adjuntamos el resumen de su informe técnico EcoSolar:\n\n" +
      "- Capacidad del Sistema: " + data.capacidadKWp + " kWp\n" +
      "- Producción Anual: " + data.produccionAnualKWh + " kWh\n" +
      "- Consumo Estimado: " + data.consumoAnualKWh + " kWh\n" +
      "- Autosuficiencia: " + data.autosuficienciaPct + "%\n" +
      "- Inversión Estimada: $" + data.inversionUSD + "\n" +
      "- Retorno de Inversión: " + data.roiAnios + " años\n\n" +
      "Notas: " + (data.observaciones || "Sin observaciones") + "\n\n" +
      "Atentamente,\nEquipo EcoSolar";
      
    MailApp.sendEmail(data.email, "Informe Técnico Solar - " + data.cliente, mensaje);
    
    return ContentService.createTextOutput(JSON.stringify({result: "success"}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({result: "error", error: err.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
